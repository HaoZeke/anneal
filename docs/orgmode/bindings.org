* Python (maturin)

The Python package =anneal= ships from PyPI as a maturin-built wheel.
The Rust extension is exposed as the submodule =anneal._core=.
The package source lives at =python/anneal/= in the repo and is installed via =maturin develop --features python= during development.

* C and C++ (cargo-c)

The Rust crate produces a =cdylib=, =staticlib=, generated header =anneal-core.h=, and pkg-config metadata =anneal-core.pc= via cargo-c.

#+BEGIN_SRC bash
cargo cinstall --release --prefix /usr/local
pkg-config --cflags --libs anneal-core
#+END_SRC

A hand-written companion header =anneal-core.hpp= wraps the C declarations in =namespace anneal= for ergonomic C++ usage.

=anneal_box_minimize= reads =ANNEAL_SUCCESS= from a callback as "keep
going". =ANNEAL_STOPPED= asks the search to return at once: the best point
found is written back, and the call itself returns =ANNEAL_STOPPED=. Any
other status is a failure and leaves the start in place. The variants live
in the generated =anneal-core.h=.

The install also places =dlpack/dlpack.h= (DLPack 1.3, the header =dlpk= 0.1.5 was translated from) on that include prefix. =anneal-core.h= includes it, so =pkg-config --cflags anneal-core= is enough to compile a consumer. Meson and CMake install the same header.

* Build-system glue

** Meson

#+BEGIN_SRC meson
anneal_core_dep = dependency('anneal-core')
executable('myprog', 'main.cpp', dependencies: anneal_core_dep)
#+END_SRC

** CMake

#+BEGIN_SRC cmake
find_package(PkgConfig REQUIRED)
pkg_check_modules(ANNEAL_CORE REQUIRED IMPORTED_TARGET anneal-core)
target_link_libraries(myprog PRIVATE PkgConfig::ANNEAL_CORE)
#+END_SRC

** Tiny C example (box ensemble + DLPack)

#+BEGIN_SRC c
#include <anneal-core.h>
#include <math.h>
#include <stdio.h>

static anneal_status_t rastrigin_val(void *user, const DLManagedTensorVersioned *x, double *out) {
    (void)user;
    const double *xs = (const double *)x->dl_tensor.data;
    int64_t d = x->dl_tensor.shape[0];
    double sum = 0.0;
    for (int64_t i = 0; i < d; ++i) {
        sum += xs[i] * xs[i] - 10.0 * cos(2.0 * M_PI * xs[i]) + 10.0;
    }
    *out = sum;
    return ANNEAL_SUCCESS;
}

static anneal_status_t rastrigin_grad(void *user, const DLManagedTensorVersioned *x,
                                      DLManagedTensorVersioned *grad_out) {
    (void)user;
    const double *xs = (const double *)x->dl_tensor.data;
    double *gs = (double *)grad_out->dl_tensor.data;
    int64_t d = x->dl_tensor.shape[0];
    for (int64_t i = 0; i < d; ++i) {
        gs[i] = 2.0 * xs[i] + 20.0 * M_PI * sin(2.0 * M_PI * xs[i]);
    }
    return ANNEAL_SUCCESS;
}

int main(void) {
    double coords[2] = {3.0, -2.0};
    double low[2] = {-5.12, -5.12};
    double high[2] = {5.12, 5.12};
    int64_t shape[1] = {2};
    DLManagedTensorVersioned x = {0};
    x.version.major = DLPACK_MAJOR_VERSION;
    x.version.minor = DLPACK_MINOR_VERSION;
    x.dl_tensor.data = coords;
    x.dl_tensor.device.device_type = kDLCPU;
    x.dl_tensor.ndim = 1;
    x.dl_tensor.dtype.code = kDLFloat;
    x.dl_tensor.dtype.bits = 64;
    x.dl_tensor.dtype.lanes = 1;
    x.dl_tensor.shape = shape;

    anneal_box_config_t cfg = {0};
    anneal_box_config_default(&cfg);
    cfg.budget = 600;
    cfg.seed = 7;

    anneal_box_report_t report = {0};
    anneal_status_t st = anneal_box_minimize(
        rastrigin_val, rastrigin_grad, NULL, &x, low, high, &cfg, &report
    );
    if (st != ANNEAL_SUCCESS && st != ANNEAL_STOPPED) {
        fprintf(stderr, "error: %s\n", anneal_last_error());
        return 1;
    }
    printf("best val %.6f at (%.4f, %.4f) in %zu evals + %zu grads\n",
           report.value, coords[0], coords[1], report.n_evals, report.n_grads);
    return 0;
}
#+END_SRC
